//

// 1. Script Properties: BOT_TOKEN = your bot token, ADMIN_CHAT_ID = your
//    Telegram numeric chat id.
// 2. Deploy as a web app (Execute as: Me, Who has access: Anyone). Copy the

// 3. Add a Script Property DEPLOY_URL = that /exec URL.
// 4. Select setWebhook in the function dropdown at the top of the editor and
//    click Run once. This tells Telegram to start sending messages here.

// Any message containing one of these skips FAQ matching and goes straight
// to a human.
var ESCALATE_KEYWORDS = ['agent', 'human', 'representative', 'real person', 'talk to someone', 'talk to support'];

function doPost(e) {
  var props = PropertiesService.getScriptProperties();
  var botToken = props.getProperty('BOT_TOKEN');
  var adminChatId = props.getProperty('ADMIN_CHAT_ID');

  if (!botToken || !adminChatId) {
    return ContentService.createTextOutput('missing config');
  }

  var update;
  try {
    update = JSON.parse(e.postData.contents);
  } catch (err) {
    return ContentService.createTextOutput('bad payload');
  }

  if (update.message) {
    handleMessage_(update.message, botToken, adminChatId);
  }

  return ContentService.createTextOutput('ok');
}

function handleMessage_(message, botToken, adminChatId) {
  var chatId = message.chat.id;
  var text = (message.text || '').trim();

  // Admin replying to a forwarded escalation -> relay the reply back to the
  // original user. This is how one admin Telegram account can answer many
  // different users without any of them seeing each other.
  if (String(chatId) === String(adminChatId) && message.reply_to_message) {
    var repliedId = message.reply_to_message.message_id;
    var cache = CacheService.getScriptCache();
    var originalChatId = cache.get('esc_' + repliedId);
    if (originalChatId) {
      sendMessage_(botToken, originalChatId, text);
      sendMessage_(botToken, adminChatId, '✅ Sent to user.');
    } else {
      sendMessage_(botToken, adminChatId, "Couldn't match this to a user (routing expires after 6 hours) — ask them to message again if it's still unresolved.");
    }
    return;
  }

  if (text === '/start') {
    sendMessage_(botToken, chatId, "👋 Welcome to NowFunded Support!\n\nAsk me about challenges, payouts, MT5 logins, KYC, discount codes, refunds, or affiliates — I'll answer instantly. If I can't help, just say \"agent\" and I'll connect you to a human.");
    return;
  }

  var lowerText = text.toLowerCase();

  var explicitEscalate = ESCALATE_KEYWORDS.some(function(k){ return lowerText.indexOf(k) !== -1; });
  if (explicitEscalate) {
    escalate_(botToken, adminChatId, message);
    return;
  }

  var faq = FAQS.find(function(f){
    return f.keywords.some(function(k){ return lowerText.indexOf(k) !== -1; });
  });

  if (faq) {
    sendMessage_(botToken, chatId, faq.answer);
  } else {
    escalate_(botToken, adminChatId, message);
  }
}

function escalate_(botToken, adminChatId, message) {
  var chatId = message.chat.id;
  var from = message.from || {};
  var name = [from.first_name, from.last_name].filter(Boolean).join(' ') || from.username || ('User ' + chatId);
  var text = message.text || '(no text)';

  var forwardText = '🆘 Support request from ' + name + (from.username ? ' (@' + from.username + ')' : '') + ':\n\n' + text + '\n\nReply to THIS message to answer them.';
  var result = sendMessage_(botToken, adminChatId, forwardText);

  if (result && result.result && result.result.message_id) {
    // 21600 seconds = 6 hours, the max CacheService allows.
    CacheService.getScriptCache().put('esc_' + result.result.message_id, String(chatId), 21600);
  }

  sendMessage_(botToken, chatId, "I've connected you to our support team — someone will reply here shortly. 🙏");
}

function sendMessage_(botToken, chatId, text) {
  var url = 'https://api.telegram.org/bot' + botToken + '/sendMessage';
  var resp = UrlFetchApp.fetch(url, {
    method: 'post',
    contentType: 'application/json',
    payload: JSON.stringify({ chat_id: chatId, text: text }),
    muteHttpExceptions: true
  });
  try {
    return JSON.parse(resp.getContentText());
  } catch (err) {
    return null;
  }
}

// Run this once manually (select it in the function dropdown at the top of
// the editor, click Run) after deploying and setting DEPLOY_URL.
function setWebhook() {
  var props = PropertiesService.getScriptProperties();
  var botToken = props.getProperty('BOT_TOKEN');
  var deployUrl = props.getProperty('DEPLOY_URL');

  if (!botToken || !deployUrl) {
    Logger.log('Set BOT_TOKEN and DEPLOY_URL script properties first.');
    return;
  }

  var url = 'https://api.telegram.org/bot' + botToken + '/setWebhook';
  var resp = UrlFetchApp.fetch(url, {
    method: 'post',
    contentType: 'application/json',
    payload: JSON.stringify({ url: deployUrl }),
    muteHttpExceptions: true
  });
  Logger.log(resp.getContentText());
}

// Optional: run manually to check webhook status / debug delivery issues.
function getWebhookInfo() {
  var botToken = PropertiesService.getScriptProperties().getProperty('BOT_TOKEN');
  var resp = UrlFetchApp.fetch('https://api.telegram.org/bot' + botToken + '/getWebhookInfo', { muteHttpExceptions: true });
  Logger.log(resp.getContentText());
}
